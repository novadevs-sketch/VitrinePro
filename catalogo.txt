        const STORAGE_KEY = 'vitrinepro_permanent_db_v2';

        function openIndexedDB() {
            return new Promise((resolve) => {
                try {
                    const req = indexedDB.open('VitrineProStorage', 1);
                    req.onupgradeneeded = (e) => {
                        const dbInst = e.target.result;
                        if (!dbInst.objectStoreNames.contains('store')) {
                            dbInst.createObjectStore('store');
                        }
                    };
                    req.onsuccess = (e) => resolve(e.target.result);
                    req.onerror = () => resolve(null);
                } catch (e) {
                    resolve(null);
                }
            });
        }

        async function loadPersistentCache() {
            try {
                const idb = await openIndexedDB();
                if (idb) {
                    const data = await new Promise((resolve) => {
                        const tx = idb.transaction('store', 'readonly');
                        const getReq = tx.objectStore('store').get(STORAGE_KEY);
                        getReq.onsuccess = () => resolve(getReq.result || null);
                        getReq.onerror = () => resolve(null);
                    });
                    if (data) return data;
                }
            } catch (e) {}

            try {
                const rawLocal = localStorage.getItem(STORAGE_KEY);
                if (rawLocal) return JSON.parse(rawLocal);
            } catch (e) {}

            try {
                if (window.name && window.name.startsWith('VITRINEPRO_DB:')) {
                    return JSON.parse(window.name.replace('VITRINEPRO_DB:', ''));
                }
                const rawSession = sessionStorage.getItem('vitrinepro_db_v1');
                if (rawSession) return JSON.parse(rawSession);
            } catch (e) {}

            return null;
        }

        async function savePersistentCache(extraProps = {}) {
            try {
                const existing = (await loadPersistentCache()) || {};
                const payload = {
                    ...existing,
                    ...extraProps,
                    segments: state.segments,
                    companies: state.companies,
                    users: state.users,
                    clicks: state.clicks,
                    adminLogs: state.adminLogs
                };

                try {
                    const idb = await openIndexedDB();
                    if (idb) {
                        await new Promise((resolve) => {
                            const tx = idb.transaction('store', 'readwrite');
                            tx.objectStore('store').put(payload, STORAGE_KEY);
                            tx.oncomplete = () => resolve(true);
                            tx.onerror = () => resolve(false);
                        });
                    }
                } catch (e) {}

                try {
                    localStorage.setItem(STORAGE_KEY, JSON.stringify(payload));
                } catch (e) {}

                try {
                    window.name = 'VITRINEPRO_DB:' + JSON.stringify(payload);
                    sessionStorage.setItem('vitrinepro_db_v1', JSON.stringify(payload));
                } catch (e) {}
            } catch (e) {}
        }

        function saveLocal(colName, docId, data) {
            if (colName === 'segments') {
                const idx = state.segments.findIndex(s => s.id === docId);
                if (idx >= 0) state.segments[idx] = data; else state.segments.push(data);
                populateSegmentDropdowns();
                renderSegmentsBar();
            } else if (colName === 'companies') {
                const idx = state.companies.findIndex(c => c.id === docId);
                if (idx >= 0) state.companies[idx] = data; else state.companies.push(data);
                renderCatalog();
                if (state.currentRoute === '/merchant') renderMerchantPanel();
                if (state.currentRoute === '/admin') renderAdminPanel();
            } else if (colName === 'users') {
                const idx = state.users.findIndex(u => u.id === docId);
                if (idx >= 0) state.users[idx] = data; else state.users.push(data);
            } else if (colName === 'clicks') {
                state.clicks.push(data);
                if (state.currentRoute === '/merchant') renderMerchantPanel();
            } else if (colName === 'admin_logs') {
                state.adminLogs.unshift(data);
            }
        }

        async function removeDocument(colName, docId) {
            if (db && firebaseReady && fbModules && auth && auth.currentUser) {
                try {
                    await fbModules.deleteDoc(fbModules.doc(db, 'artifacts', appId, 'public', 'data', colName, docId));
                } catch(e) {}
            }
            if (colName === 'companies') state.companies = state.companies.filter(c => c.id !== docId);
            if (colName === 'segments') state.segments = state.segments.filter(s => s.id !== docId);
            await savePersistentCache();
            renderCatalog();
            if (state.currentRoute === '/admin') renderAdminPanel();
        }

        function renderApp() {
            const views = ['view-catalog', 'view-digital-card', 'view-auth', 'view-merchant', 'view-admin'];
            views.forEach(id => {
                const el = document.getElementById(id);
                if (el) el.classList.add('hidden');
            });

            renderHeader();

            if (state.currentRoute === '/') {
                document.getElementById('view-catalog').classList.remove('hidden');
                renderSegmentsBar();
                renderCatalog();
            } else if (state.currentRoute.startsWith('/empresa/')) {
                const slug = state.currentRoute.replace('/empresa/', '');
                document.getElementById('view-digital-card').classList.remove('hidden');
                renderDigitalCard(slug);
            } else if (state.currentRoute === '/auth') {
                document.getElementById('view-auth').classList.remove('hidden');
                populateSegmentDropdowns();
            } else if (state.currentRoute === '/merchant') {
                if (!state.currentUser || state.currentUser.role !== 'merchant') {
                    navigateTo('/auth');
                    return;
                }
                document.getElementById('view-merchant').classList.remove('hidden');
                renderMerchantPanel();
            } else if (state.currentRoute === '/admin') {
                if (!state.currentUser || state.currentUser.role !== 'admin') {
                    navigateTo('/auth');
                    return;
                }
                document.getElementById('view-admin').classList.remove('hidden');
                renderAdminPanel();
            } else {
                document.getElementById('view-catalog').classList.remove('hidden');
                renderCatalog();
            }
        }

        window.navigateTo = function(route) {
            state.currentRoute = route;
            try { window.location.hash = route; } catch(e) {}
            window.scrollTo({ top: 0, behavior: 'smooth' });
            renderApp();
        };

        async function loadUFs() {
            const ufSelect = document.getElementById('reg-state-select');
            if (!ufSelect || ufSelect.options.length > 1) return;

            let statesList = FALLBACK_STATES;
            try {
                const res = await fetch('https://servicodados.ibge.gov.br/api/v1/localidades/estados?orderBy=nome');
                if (res.ok) {
                    statesList = await res.json();
                }
            } catch (e) {}

            ufSelect.innerHTML = '<option value="">Selecione UF</option>' +
                statesList.map(uf => `<option value="${uf.sigla}">${uf.sigla} - ${uf.nome}</option>`).join('');
        }

        window.onStateSelectChange = async function() {
            const uf = document.getElementById('reg-state-select').value;
            const citySelect = document.getElementById('reg-city-select');
            if (!uf) {
                citySelect.innerHTML = '<option value="">Cidade</option>';
                citySelect.disabled = true;
                return;
            }

            citySelect.disabled = true;
            citySelect.innerHTML = '<option value="">Carregando...</option>';

            try {
                const res = await fetch(`https://servicodados.ibge.gov.br/api/v1/localidades/estados/${uf}/municipios?orderBy=nome`);
                if (res.ok) {
                    const cities = await res.json();
                    citySelect.innerHTML = '<option value="">Selecione a cidade</option>' +
                        cities.map(c => `<option value="${c.nome}">${c.nome}</option>`).join('');
                    citySelect.disabled = false;
                    return;
                }
            } catch (e) {}

            if (uf === 'TO') {
                citySelect.innerHTML = '<option value="">Selecione a cidade</option>' +
                    FALLBACK_CITIES_TO.map(c => `<option value="${c}">${c}</option>`).join('');
                citySelect.disabled = false;
            } else {
                toggleCityInputMode();
                document.getElementById('reg-city').placeholder = `Digite sua cidade - ${uf}`;
                showToast('Digite o nome da sua cidade no campo.', 'info');
            }
        };

        function normalizeText(str) {
            return (str || '').toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '').trim();
        }

        function renderCatalog() {
            const grid = document.getElementById('companies-grid');
            const empty = document.getElementById('catalog-empty');
            const title = document.getElementById('catalog-section-title');
            const countLabel = document.getElementById('catalog-count-label');
            if (!grid) return;

            const isServiceTab = state.catalogCategory === 'service';
            const hasSearch = state.searchQuery && state.searchQuery.length > 0;

            let list = state.companies.filter(c => {
                if (c.status !== 'approved') return false;
                // Quando houver termo de busca, pesquisa tanto em Empresas quanto em Prestadores de Serviços
                if (hasSearch) return true;
                const type = c.entity_type || 'commerce';
                return isServiceTab ? type === 'service' : type !== 'service';
            });

            if (state.selectedSegment !== 'all' && !hasSearch) {
                list = list.filter(c => c.segment === state.selectedSegment);
                if (title) title.textContent = state.selectedSegment;
            } else if (hasSearch) {
                if (title) title.textContent = 'Resultados da Busca (Empresas e Serviços)';
            } else {
                if (title) title.textContent = isServiceTab ? 'Todos os Prestadores de Serviços' : 'Todas as Empresas';
            }

            if (hasSearch) {
                list = list.filter(c =>
                    normalizeText(c.name).includes(state.searchQuery) ||
                    normalizeText(c.segment).includes(state.searchQuery) ||
                    normalizeText(c.city).includes(state.searchQuery) ||
                    normalizeText(c.description).includes(state.searchQuery)
                );
            }

            if (state.sortBy === 'name_asc') {
                list.sort((a, b) => a.name.localeCompare(b.name));
            } else if (state.sortBy === 'name_desc') {
                list.sort((a, b) => b.name.localeCompare(a.name));
            } else {
                list.sort((a, b) => (b.order_active ? 1 : 0) - (a.order_active ? 1 : 0));
            }

            if (countLabel) {
                if (hasSearch) {
                    countLabel.textContent = `${list.length} ${list.length === 1 ? 'resultado encontrado' : 'resultados encontrados'}`;
                } else {
                    const nounSingular = isServiceTab ? 'prestador encontrado' : 'empresa encontrada';
                    const nounPlural = isServiceTab ? 'prestadores encontrados' : 'empresas encontradas';
                    countLabel.textContent = `${list.length} ${list.length === 1 ? nounSingular : nounPlural}`;
                }
            }

            if (list.length === 0) {
                grid.innerHTML = '';
                if (empty) empty.classList.remove('hidden');
                return;
            }

            if (empty) empty.classList.add('hidden');

            grid.innerHTML = list.map(comp => {

                const hasApprovedOrder = comp.order_active === true && comp.order_url && comp.order_url.trim() !== '';

                return `
                    <article class="bg-white rounded-3xl border border-slate-200/80 shadow-slim hover:shadow-card-hover transition-all duration-300 flex flex-col overflow-hidden group">
                        <div class="p-5 flex-grow flex flex-col">
                            <div class="flex items-start gap-3.5 mb-3.5">
                                <img src="${comp.logo_url}" onerror="this.src='https://placehold.co/120x120/eef2ff/4f46e5?text=PERFIL'" alt="${comp.name}" class="w-14 h-14 rounded-2xl object-cover border border-slate-100 shrink-0 group-hover:scale-105 transition-transform">
                                <div class="min-w-0 flex-grow">
                                    <span class="inline-block px-2.5 py-0.5 rounded-lg text-[10px] font-bold uppercase tracking-wider bg-brand-50 text-brand-700 mb-1 truncate max-w-full">${comp.segment || 'Prestador de Serviços'}</span>
                                    <h4 class="font-bold text-base text-slate-900 leading-snug truncate cursor-pointer hover:text-brand-600 transition" onclick="navigateTo('/empresa/${comp.slug}')">${comp.name}</h4>
                                    <p class="text-xs text-slate-400 flex items-center gap-1 mt-0.5 truncate">
                                        <i class="ph-fill ph-map-pin text-slate-400"></i> ${comp.city}
                                    </p>
                                </div>
                            </div>

                            <p class="text-xs text-slate-500 line-clamp-2 mb-4 flex-grow leading-relaxed">${comp.description}</p>

                            <div class="pt-3 border-t border-slate-100 space-y-2.5">
                                <div class="grid ${hasIg ? 'grid-cols-3' : 'grid-cols-2'} gap-2">
                                    <a href="${waUrl}" target="_blank" onclick="registerClick('${comp.id}', 'whatsapp')" title="WhatsApp" class="py-2 px-2 rounded-xl bg-emerald-50 hover:bg-emerald-600 text-emerald-700 hover:text-white font-semibold text-xs flex items-center justify-center gap-1.5 transition">

        window.handleImageUpload = function(event, previewId, dataId) {
            const file = event.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = function(e) {
                const img = new Image();
                img.onload = function() {
                    const canvas = document.createElement('canvas');
                    const maxDim = 320;
                    let w = img.width;
                    let h = img.height;
                    if (w > maxDim || h > maxDim) {
                        if (w > h) { h = Math.round((h * maxDim) / w); w = maxDim; }
                        else { w = Math.round((w * maxDim) / h); h = maxDim; }
                    }
                    canvas.width = w;
                    canvas.height = h;
                    const ctx = canvas.getContext('2d');
                    ctx.drawImage(img, 0, 0, w, h);
                    const compressed = canvas.toDataURL('image/jpeg', 0.8);
                    document.getElementById(previewId).src = compressed;
                    document.getElementById(dataId).value = compressed;
                };
                img.src = e.target.result;
            };
            reader.readAsDataURL(file);
        };

        window.handleSaveCloudConfig = async function(e) {
            e.preventDefault();
            const raw = document.getElementById('cloud-config-json').value.trim();
            try {
                const jsonStart = raw.indexOf('{');
                const jsonEnd = raw.lastIndexOf('}');
                const cleanJson = raw.substring(jsonStart, jsonEnd + 1);
                const parsed = JSON.parse(cleanJson);
                
                await savePersistentCache({ firebaseConfig: parsed });

                const ok = await connectCloudDatabase(parsed);
                if (ok) {
                    state.cloudConnected = true;
                    for (const seg of state.segments) await saveDocument('segments', seg.id, seg);
                    for (const comp of state.companies) await saveDocument('companies', comp.id, comp);
                    for (const usr of state.users) await saveDocument('users', usr.id, usr);
                    showToast('Banco de dados na nuvem conectado e salvo permanentemente!');
                    closeModal();
                    renderAdminPanel();
                } else {
                    showToast('Não foi possível conectar. Verifique as chaves informadas.', 'error');
                }
            } catch (err) {
                showToast('Formato JSON inválido. Cole apenas o objeto { ... }.', 'error');
            }
        };

            if (hash !== state.currentRoute) {
                state.currentRoute = hash;
                renderApp();
            }
        });
